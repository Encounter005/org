* 切片和索引
切片和索引是通过对已有的array进行操作从而得到想要的部分元素的过程， =start:stop:step=
核心是把处理按维度分开，不处理的维度统一用:或...代替，在看操作时，也要首先关注“，”在哪
+ 注意：索引支持负数（从后往前索引）
#+begin_src python :results output
import numpy as np
rng = np.random.default_rng(seed=42)
arr = rng.integers(0, 20, (5, 4))
print(f"arr is:\n {arr}")
# NOTE: use [] to get row
print(f"arr[0] is:\n {arr[0]}")
print(f"row0 to row3 arr[0:3] is:\n {arr[0:3]}")
print(f"row0 and row 3 arr[[0, 3]] is:\n {arr[[0, 3]]}")
print(f"row1 to row3 and col1 arr[1:3, 1] is:\n {arr[1:3, 1]}")
print(f"row1, row3, col0 arr[[1, 3], 1] is:\n {arr[[1, 3], 0]}")
print(f"row3 to end, arr[3:] is {arr[3:]}")
print(f"row0 to row3, col1 to col3, arr[:3, 1:3] is:\n {arr[:3, 1:3]}")
# NOTE: start:stop:step
print(f"row1 to row4, step is 2 arr[1:4:2] is:\n {arr[1:4:2]}")
print(f"row1 to row4, step is 2, col0 and col2 arr[1:4:2, 0:3:2] is:\n {arr[1:4:2, 0:3:2]}")
print(f"all value of col1  arr[..., 1] or arr[:, 1] is:\n {arr[:, 1]}")
#+end_src

* 拼接
有时候我们需要对已有的几个array进行拼接形成一个大的array（例如不同类型特征的拼接）， =np.concatenate（拼接）= 和 =np.stack(堆叠)=
#+begin_src python :results output
import numpy as np

rng = np.random.default_rng(seed=42)
arr1 = rng.random((2, 3))
arr2 = rng.random((2, 3))
print(f"arr1 is:\n {arr1}")
print(f"arr2 is:\n {arr2}")

# NOTE: 默认axis=0（列）连接
print(f"concatenate arr1, arr2 by col is:\n {np.concatenate((arr1, arr2))}")
print(f"concatenate arr1, arr2 by row is:\n {np.concatenate((arr1, arr2), axis=1)}")

# NOTE: 堆叠，默认根据axis=0进行
stack1 = np.stack((arr1, arr2))
print(f"stack arr1, arr2 by col is:\n {stack1}\n shape is: {stack1.shape}")
stack2 = np.stack((arr1, arr2), axis=2)
print(f"stack arr1, arr2, axis = 2 is:\n {stack2}\n shape is: {stack2.shape}")
#+end_src
* 重复
重复是另一种拼接方式，也可以指定要重复的维度
+ 注意：重复的过程是一个维度一个维度重复，而不是整个array重复
#+begin_src python :results output
import numpy as np

rng = np.random.default_rng(seed=42)
arr = rng.integers(0, 10, (3, 4))
print(f"arr is:\n {arr}")

# NOTE: 在列重复2次
print(f"repeat twice by col arr is:\n {np.repeat(arr, 2 , axis=0)}")
print(f"repeat 3 times by row arr is:\n {np.repeat(arr, 3, axis=1)}")
#+end_src
